import csv import random # 定義一些基本的隨機母體資料 first_names = ['偉', '芳', '娜', '敏', '強', '麗', '勇', '秀英', '洋', '靜', '豪', '婷'] last_names = ['張', '王', '李', '陳', '林', '黃', '吳', '蔡', '楊', '許', '鄭', '謝'] cities = ['台北市', '台中市', '高雄市', '台南市', '新竹市', '桃園市'] data = [] # 迴圈產生 600 筆資料 for i in range(1, 601): user_id = i name = random.choice(last_names) + random.choice(first_names) age = random.randint(18, 65) city = random.choice(cities) score = random.randint(60, 100) # 每筆資料的格式 [ID, 姓名, 年齡, 城市, 分數] data.append([user_id, name, age, city, score]) # 寫入 CSV 檔案 filename = '600_records.csv' with open(filename, 'w', newline='', encoding='utf-8-sig') as f: writer = csv.writer(f) # 寫入表頭 writer.writerow(['ID', '姓名', '年齡', '城市', '分數']) # 寫入 600 筆資料 writer.writerows(data) print(f'已成功產生 "{filename}"，共 600 筆資料！')
